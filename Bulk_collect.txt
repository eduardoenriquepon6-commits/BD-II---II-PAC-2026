DECLARE 

    TYPE REGISTRO_CATEGORIAS IS RECORD(

        ID_CAT CATEGORIAS.CATEGORIAID%TYPE,
        NOM_CAT CATEGORIAS.NOMBRECAT%TYPE

    );

    TYPE TABLA_CATEGORIAS IS TABLE OF REGISTRO_CATEGORIAS;

    DATOS_CATEGORIAS TABLA_CATEGORIAS;

    TYPE REGISTRO_PROVEEDORES IS RECORD(

        ID_PROV PROVEEDORES.PROVEEDORID%TYPE,
        NOM_PROV PROVEEDORES.NOMBREPROV%TYPE,
        CONTACTO_PROV PROVEEDORES.CONTACTO%TYPE,
        CEL_PROV PROVEEDORES.CELUPROV%TYPE,
        FIJO_PROV PROVEEDORES.FIJOPROV%TYPE

    );

    TYPE TABLA_PROVEEDORES IS TABLE OF REGISTRO_PROVEEDORES;

    DATOS_PROVEEDORES TABLA_PROVEEDORES;

    TYPE REGISTRO_PRODUCTOS IS RECORD(

        ID_PROD PRODUCTOS.PRODUCTOID%TYPE,
        ID_PROV PRODUCTOS.PROVEEDORID%TYPE,
        ID_CAT PRODUCTOS.CATEGORIAID%TYPE,
        DESCRIPCION PRODUCTOS.DESCRIPCION%TYPE,
        PRECIO_UNITARIO PRODUCTOS.PRECIOUNIT%TYPE,
        EXISTENCIA PRODUCTOS.EXISTENCIA%TYPE

    );

    TYPE TABLA_PRODUCTOS IS TABLE OF REGISTRO_PRODUCTOS;

    DATOS_PRODUCTOS TABLA_PRODUCTOS;

BEGIN

    SELECT *
    BULK COLLECT INTO DATOS_CATEGORIAS
    FROM C##_PEDIDOS.CATEGORIAS;

    FORALL NUMFILA IN 1..DATOS_CATEGORIAS.COUNT

        INSERT INTO CATEGORIAS VALUES (

            DATOS_CATEGORIAS(NUMFILA).ID_CAT, 
            DATOS_CATEGORIAS(NUMFILA).NOM_CAT

        );

    DBMS_OUTPUT.PUT_LINE(DATOS_CATEGORIAS.COUNT || 'REGISTROS INSERTADOS EN CATEGORIAS');

    SELECT *
    BULK COLLECT INTO DATOS_PROVEEDORES
    FROM C##_PEDIDOS.PROVEEDORES;

    FORALL NUMFILA IN 1..DATOS_PROVEEDORES.COUNT

        INSERT INTO PROVEEDORES VALUES (

            DATOS_PROVEEDORES(NUMFILA).ID_PROV,
            DATOS_PROVEEDORES(NUMFILA).NOM_PROV,
            DATOS_PROVEEDORES(NUMFILA).CONTACTO_PROV,
            DATOS_PROVEEDORES(NUMFILA).CEL_PROV,
            DATOS_PROVEEDORES(NUMFILA).FIJO_PROV

        );
        COMMIT;

    DBMS_OUTPUT.PUT_LINE(DATOS_PROVEEDORES.COUNT || 'REGISTROS INSERTADOS EN PROVEEDORES');

    SELECT *
    BULK COLLECT INTO DATOS_PRODUCTOS
    FROM C##_PEDIDOS.PRODUCTOS;

    FORALL NUMFILA IN 1..DATOS_PRODUCTOS.COUNT

        INSERT INTO PRODUCTOS VALUES (

            DATOS_PRODUCTOS(NUMFILA).ID_PROD,
            DATOS_PRODUCTOS(NUMFILA).ID_PROV,
            DATOS_PRODUCTOS(NUMFILA).ID_CAT,
            DATOS_PRODUCTOS(NUMFILA).DESCRIPCION,
            DATOS_PRODUCTOS(NUMFILA).PRECIO_UNITARIO,
            DATOS_PRODUCTOS(NUMFILA).EXISTENCIA

        );
        COMMIT;

    DBMS_OUTPUT.PUT_LINE(DATOS_PRODUCTOS.COUNT || 'REGISTROS INSERTADOS EN PRODUCTOS');

    EXCEPTION

        WHEN OTHERS THEN

            IF (SQLCODE = -1) THEN

                DBMS_OUTPUT.PUT_LINE('REGISTRO DUPLICADO');
                DBMS_OUTPUT.PUT_LINE(SQLCODE);
            
            END IF;

END;
